# 网页界面
streamlit>=1.28.0

# 数值与音频
# numpy / scipy 必须成对限制上限，否则「切韵对照」会静默失效：
#   qieyun 0.13 -> qieyun-encoder -> networkx<2.6（实际 2.5.1）仍在用 np.float_，
#   该属性在 numpy 2.0 已移除，导入 Qieyun 会抛 AttributeError；
#   而 scipy>=1.14 又要求 numpy>=2，两者无法同时满足无上限安装。
# 现固定为 numpy 1.26.x + scipy 1.13.x，已验证 Qieyun 可正常导入。
numpy>=1.24.0,<2.0
scipy>=1.10.0,<1.14

# pyarrow 是 streamlit 的传递依赖（约束 >=10,<27），但 18 起要求 numpy>=2，
# 与上面的 numpy<2 冲突：装到 26 会让 robot/app.py 的「按住说话」自定义组件
# 直接报 "pyarrow requires NumPy 2.0 or newer"，整块录音控件不可用。
# 17.0.0 是仍支持 numpy 1.26 的版本，已验证与 Qieyun 共存。
pyarrow>=10,<18
